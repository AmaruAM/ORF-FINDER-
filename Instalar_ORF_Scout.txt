#!/usr/bin/env bash
# Instalación en Linux o Google Cloud Shell. Copia esta línea completa en tu terminal:
# wget -O Instalar_ORF_Scout.txt https://raw.githubusercontent.com/AmaruAM/ORF-FINDER-/main/Instalar_ORF_Scout.txt && bash Instalar_ORF_Scout.txt
set -euo pipefail
dest=${1:-"$HOME/ORF_Scout"}
for tool in wget unzip sha256sum install; do
  command -v "$tool" >/dev/null || { echo "Falta $tool en este equipo." >&2; exit 1; }
done
tmp=$(mktemp -d)
trap 'rm -rf -- "$tmp"' EXIT
url=https://raw.githubusercontent.com/AmaruAM/ORF-FINDER-/main/ORF_Scout.zip
wget -nv -O "$tmp/ORF_Scout.zip" "$url"
(cd "$tmp" && printf '%s  ORF_Scout.zip\n' 'a4285f21f33716d84742904718d40783401139ba572b1ecf9bdfeb4cfb1d72be' | sha256sum -c - && unzip -q ORF_Scout.zip)
mkdir -p -- "$dest"/{scripts/catalogo,guias,corridas}
for script in orf_scout.sh starvation_scout.sh; do
  install -m 755 -- "$tmp/ORF_Scout/$script" "$dest/scripts/$script"
done
install -m 644 -- "$tmp/ORF_Scout/catalogo/marcadores.tsv" "$dest/scripts/catalogo/marcadores.tsv"
install -m 644 -- "$tmp/ORF_Scout/catalogo/referencias.tsv" "$dest/scripts/catalogo/referencias.tsv"
install -m 644 -- "$tmp/ORF_Scout/"*.md "$dest/guias/"
printf '\nInstalación lista en %s\n' "$dest"
printf 'Para iniciar una corrida: %q\n' "$dest/scripts/orf_scout.sh"
